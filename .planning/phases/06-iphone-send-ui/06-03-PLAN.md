```typescript
import { resizeImageToBase64, insertAtCursor } from './page';

describe('SEND-03: 画像添付', () => {
  it('resizeImageToBase64 が Canvas API を使って base64 文字列を返す', async () => {
    // モック設定: HTMLCanvasElement.prototype.getContext は beforeEach で設定済み
    // URL.createObjectURL / URL.revokeObjectURL モック
    const originalCreateObjectURL = URL.createObjectURL;
    const originalRevokeObjectURL = URL.revokeObjectURL;
    URL.createObjectURL = vi.fn().mockReturnValue('blob:mock');
    URL.revokeObjectURL = vi.fn();

    // Image の onload を手動トリガーするモック
    const originalImage = global.Image;
    (global as any).Image = class {
      width = 1600;
      height = 1200;
      set src(_: string) {
        // onload を同期的にトリガー
        setTimeout(() => (this as any).onload?.(), 0);
      }
    };

    const file = new File([''], 'test.jpg', { type: 'image/jpeg' });
    const result = await resizeImageToBase64(file, 800);

    // Canvas の toDataURL が呼ばれていること
    expect(result).toMatch(/^data:image\/jpeg/);

    // クリーンアップ
    URL.createObjectURL = originalCreateObjectURL;
    URL.revokeObjectURL = originalRevokeObjectURL;
    (global as any).Image = originalImage;
  });

  it('insertAtCursor がカーソル位置に文字列を挿入して新しい value を返す', () => {
    const textarea = document.createElement('textarea');
    textarea.value = 'hello world';
    textarea.selectionStart = 5;
    textarea.selectionEnd = 5;

    const result = insertAtCursor(textarea, '![](data:image/jpeg;base64,abc)');
    expect(result).toBe('hello![](data:image/jpeg;base64,abc) world');
  });
});
```

**注意:** page.tsx は 'use client' があるが vitest の jsdom 環境ではサーバー側 import は発生しない。import 自体は問題ない。ただし page.tsx を import すると useEffect 等が実行されるため、`vi.mock` で副作用をブロックすることが必要な場合は、関数のみを named import でテストする。

もし page.tsx import が問題になる場合（localStorage 未定義等）: `vi.stubGlobal('localStorage', { getItem: vi.fn(), setItem: vi.fn(), removeItem: vi.fn() })` を beforeEach に追加する。
