**モジュール構成:**

```rust
// src-tauri/src/gdrive.rs

use std::sync::Mutex;
use std::path::PathBuf;
use reqwest::Client;
use serde::{Deserialize, Serialize};
use directories::BaseDirs;
use crate::state::{AppState, ProConfig};

// ------ 定数 ------
const FOLDER_NAME: &str = "ore-no-fusen";
const PUSH_CONFIG_FILE: &str = "fusen_push_config.json";
const NOTE_FILE: &str = "fusen_note.json";
const TOKEN_FILE: &str = "gdrive_token.json";

// ------ 型定義 ------
#[derive(Serialize, Deserialize, Clone)]
pub struct SavedToken {
    pub refresh_token: String,
    pub access_token: Option<String>,
    pub expires_at: Option<i64>,  // Unix timestamp
}

#[derive(Deserialize)]
struct DriveFileList {
    files: Vec<DriveFile>,
}

#[derive(Deserialize, Clone)]
struct DriveFile {
    id: String,
}

#[derive(Deserialize)]
struct PushConfigJson {
    endpoint: String,
    keys: PushConfigKeys,
}

#[derive(Deserialize)]
struct PushConfigKeys {
    p256dh: String,
    auth: String,
}
```

**実装する公開関数:**

1. `pub fn get_token_path() -> PathBuf`
   - `directories::BaseDirs::new()` で app data dir を取得
   - `{data_local_dir}/ore-no-fusen/{TOKEN_FILE}` を返す
   - ディレクトリが存在しない場合は `create_dir_all` で作成

2. `pub async fn oauth_pkce_flow(app: &tauri::AppHandle) -> Result<SavedToken, String>`
   - oauth2 v5 の `BasicClient` で PKCE フロー
   - `tauri_plugin_oauth::start()` でローカルサーバー起動
   - ブラウザで認証 URL を開く（`tauri_plugin_shell` または `std::process::Command::new("cmd").args(["/c", "start", url])` でブラウザを起動）
   - コールバックで code を受け取り token exchange
   - SavedToken を `get_token_path()` に JSON で保存
   - Client ID / Secret は Google Cloud Console で取得したものをハードコードせず、環境変数 `GDRIVE_CLIENT_ID` / `GDRIVE_CLIENT_SECRET` から読む（設定ファイルは Phase 5 以降）
   - **注意**: tauri-plugin-oauth の callback URL は `http://127.0.0.1:{port}` 形式

3. `pub async fn get_access_token(client: &Client) -> Result<String, String>`
   - `get_token_path()` から SavedToken を読み込む
   - `expires_at` が現在時刻+60秒以内なら refresh
   - Google OAuth2 token endpoint `https://oauth2.googleapis.com/token` に POST して access_token を取得
   - 取得した access_token + 新しい expires_at を SavedToken に上書き保存して返す
   - ファイルが存在しない場合は `Err("OAuth not configured. Run fusen_oauth_connect first.".to_string())`

4. `pub async fn ensure_folder(client: &Client, access_token: &str) -> Result<String, String>`
   - Google Drive API v3 で `name='ore-no-fusen' and mimeType='application/vnd.google-apps.folder' and trashed=false` でフォルダを検索
   - 存在しない場合は作成
   - フォルダの file id を返す

5. `pub async fn upload_json(client: &Client, access_token: &str, filename: &str, data: &serde_json::Value) -> Result<(), String>`
   - `ensure_folder()` でフォルダ id を取得
   - filename で既存ファイルを検索（Drive list API）
   - 存在する場合は PATCH（上書き更新）、存在しない場合は multipart upload（POST）
   - metadata: `{ "name": filename, "parents": [folder_id] }`

6. `pub async fn download_json(client: &Client, access_token: &str, filename: &str) -> Result<serde_json::Value, String>`
   - `ensure_folder()` でフォルダ id を取得
   - filename で file id を検索
   - `https://www.googleapis.com/drive/v3/files/{id}?alt=media` で内容を取得
   - serde_json::Value として返す

7. `pub async fn poll_push_config(client: &Client, state: &Mutex<AppState>) -> Result<(), String>`
   - `get_access_token(client).await?` でトークン取得
   - `download_json(client, &token, PUSH_CONFIG_FILE).await?` で push_config を取得
   - JSON を PushConfigJson にパースして ProConfig に変換
   - `state.lock()` して `AppState.pro_config = Some(pro_config)` を設定

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_get_token_path_returns_valid_path() {
        let path = get_token_path();
        assert!(path.to_string_lossy().contains("ore-no-fusen"));
        assert!(path.to_string_lossy().ends_with("gdrive_token.json"));
    }

    // upload/download/poll は reqwest を使うためモック化が困難。
    // 構造体のパース・変換ロジックのみ単体テスト対象とする。
    #[test]
    fn test_push_config_json_parses_to_pro_config() {
        let json_str = r#"{"endpoint":"https://api.push.apple.com/3/device/ABC","keys":{"p256dh":"BNcR","auth":"tBy8"},"created_at":"2026-01-01T00:00:00Z"}"#;
        let config: PushConfigJson = serde_json::from_str(json_str).unwrap();
        assert_eq!(config.endpoint, "https://api.push.apple.com/3/device/ABC");
        assert_eq!(config.keys.p256dh, "BNcR");
        assert_eq!(config.keys.auth, "tBy8");
    }
}
